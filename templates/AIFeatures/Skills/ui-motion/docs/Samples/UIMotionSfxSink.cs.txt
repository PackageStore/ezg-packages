using Assets._Project.Core.Modules.UIMotion;

namespace MyGame.Audio // TODO: namespace của project
{
    public sealed class ProjectUIMotionSfxSink : IUIMotionSfxSink // TODO: tên class
    {
        // Constructor rỗng: module tự tạo sink theo tên class ghi trong Settings.
        public ProjectUIMotionSfxSink() { }

        public void Play(UIMotionSfxRequest request)
        {
            // request.Key = key trong RoleMap hoặc "ui.<role>.<event>" (vd "ui.button.click"); request.Volume, request.Intensity
            // (0..1) dùng nếu audio manager nhận âm lượng.
            switch (request.Event)
            {
                case UIMotionEvent.Click:
                    if (IsPressable(request.Role)) Play("TODO: tiếng bấm nút của project");
                    break;
                case UIMotionEvent.Show:
                    if (IsScreenLike(request.Role)) Play("TODO: tiếng mở popup / screen");
                    break;
                case UIMotionEvent.Hide:
                    if (IsScreenLike(request.Role)) Play("TODO: tiếng đóng popup / screen");
                    break;
                case UIMotionEvent.Arrive:
                    Play("TODO: tiếng coin / tài nguyên bay tới đích");
                    break;
                case UIMotionEvent.Error:
                    Play("TODO: tiếng lỗi (tuỳ chọn)");
                    break;
            }
        }

        private static void Play(string sound)
        {
            // TODO: lệnh phát của project, vd AudioManager.Instance.PlaySfx(sound);
        }

        private static bool IsPressable(UIRole role)
        {
            return role == UIRole.Button || role == UIRole.Toggle || role == UIRole.Tab || role == UIRole.Card ||
                   role == UIRole.Dropdown;
        }

        private static bool IsScreenLike(UIRole role)
        {
            return role == UIRole.Screen || role == UIRole.Popup || role == UIRole.Sheet;
        }
    }
}
